{
  description = "Instance registry dev shell";

  inputs = {
    nixpkgs.url = "github:NixOS/nixpkgs/nixos-unstable";
    flake-utils.url = "github:numtide/flake-utils";
  };

  outputs = { self, nixpkgs, flake-utils, ... }:
    flake-utils.lib.eachDefaultSystem (system:
      let
        pkgs = import nixpkgs { inherit system; };
        version = "0.3.1";
        # Tools every contributor needs
        tools = with pkgs; [ cargo-nextest jq nodejs_24 ];
      in
      {
        packages.default = pkgs.rustPlatform.buildRustPackage {
          pname = "instance-registry";
          inherit version;
          src = ./.;
          cargoLock.lockFile = ./Cargo.lock;
          doCheck = system != "aarch64-darwin";
        };

        devShells.default = pkgs.mkShell {
          packages = tools ++ [ pkgs.rustc pkgs.cargo ];
          shellHook = ''
            echo "registry ${version} dev shell"
            export RUST_LOG=''${RUST_LOG:-info}
          '';
        };
      });
}
